互斥锁可以用于解决优先级反转的问题，避免高优先级任务请求资源时，资源正好被低优先级任务上锁，结果此时中优先级任务触发，导致高优先级任务一直无法执行，也就是优先级反转的问题


下面是个基于ESP32 IDF的例子，用代码展示优先级反转和继承的场景。

```c
#include <stdio.h>
#include <freertos/FreeRTOS.h>
#include <freertos/task.h>
#include <freertos/semphr.h>

// 互斥锁句柄
SemaphoreHandle_t g_mutex;

// 高优先级任务A：模拟实时处理
void task_a(void *param) {
    printf("任务A启动，优先级最高，准备抢锁\n");
    
    // 尝试获取互斥锁，无限等待
    if (xSemaphoreTake(g_mutex, portMAX_DELAY) == pdTRUE) {
        printf("任务A抢到锁，开始处理关键资源\n");
        vTaskDelay(pdMS_TO_TICKS(500)); // 模拟干活，比如处理传感器数据
        printf("任务A干完活，释放锁\n");
        xSemaphoreGive(g_mutex);
    }
    
    printf("任务A完工，退出\n");
    vTaskDelete(NULL);
}

// 中优先级任务B：模拟后台任务
void task_b(void *param) {
    printf("任务B启动，优先级中等，干点杂活\n");
    
    // 模拟一些不涉及锁的后台工作
    for (int i = 0; i < 10; i++) {
        printf("任务B在忙，处理第%d轮杂活\n", i + 1);
        vTaskDelay(pdMS_TO_TICKS(300)); // 模拟耗时操作
    }
    
    printf("任务B完工，退出\n");
    vTaskDelete(NULL);
}

// 低优先级任务C：模拟访问共享资源
void task_c(void *param) {
    printf("任务C启动，优先级最低，准备占锁\n");
    
    // 抢占互斥锁
    xSemaphoreTake(g_mutex, portMAX_DELAY);
    printf("任务C拿到锁，慢悠悠处理资源\n");
    vTaskDelay(pdMS_TO_TICKS(2000)); // 模拟长时间占用资源
    printf("任务C处理完，释放锁\n");
    xSemaphoreGive(g_mutex);
    
    printf("任务C完工，退出\n");
    vTaskDelete(NULL);
}

// 主函数，初始化任务和互斥锁
void app_main(void) {
    // 创建互斥锁，默认支持优先级继承
    g_mutex = xSemaphoreCreateMutex();
    if (g_mutex == NULL) {
        printf("创建互斥锁失败，检查下内存吧\n");
        return;
    }
    
    // 创建低优先级任务C
    xTaskCreate(task_c, "TaskC", 2048, NULL, 1, NULL);
    
    // 稍微延迟，确保任务C先跑起来
    vTaskDelay(pdMS_TO_TICKS(100));
    
    // 创建高优先级任务A
    xTaskCreate(task_a, "TaskA", 2048, NULL, 3, NULL);
    
    // 创建中优先级任务B
    xTaskCreate(task_b, "TaskB", 2048, NULL, 2, NULL);
}
```

### 代码解析

1. 互斥锁创建：用 xSemaphoreCreateMutex() 创建互斥锁，默认开启优先级继承。如果任务C占着锁，任务A等着，任务C的优先级会临时升到和任务A一样（优先级3），防止被任务B抢占。
2. 任务行为：
* 任务C先抢到锁，模拟长时间操作（比如写Flash）。
* 任务A想抢锁但被卡住，触发优先级继承。
* 任务B虽然优先级比任务C高，但因为任务C被提了优先级，它只能等着。

实际效果：任务C会尽快完成，释放锁给任务A，保证高优先级任务不被耽误。

跑这段代码，你会看到任务A能快速拿到锁继续执行，而不是被任务B卡住。这就是优先级继承的威力。
