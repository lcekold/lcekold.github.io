# 程序功能

* 生产者任务：每2秒发送一个信号量
* 消费者任务：等待信号量，收到后处理数据

# 代码例程

```c
#include "FreeRTOS.h"
#include "task.h"
#include "semphr.h"
#include "stdio.h"

// 全局信号量句柄
SemaphoreHandle_t xBinarySemaphore;

// 生产者任务 - 发送信号量
void vProducerTask(void *pvParameters)
{
    TickType_t xLastWakeTime;
    const TickType_t xFrequency = pdMS_TO_TICKS(2000); // 每2秒执行一次
    
    // 初始化xLastWakeTime变量
    xLastWakeTime = xTaskGetTickCount();
    
    while(1)
    {
        // 等待2秒
        vTaskDelayUntil(&xLastWakeTime, xFrequency);
        
        printf("生产者: 准备发送信号量...\n");
        
        // 释放信号量，通知消费者任务
        if(xSemaphoreGive(xBinarySemaphore) == pdTRUE)
        {
            printf("生产者: 信号量发送成功！\n");
        }
        else
        {
            printf("生产者: 信号量发送失败！\n");
        }
    }
}

// 消费者任务 - 接收信号量
void vConsumerTask(void *pvParameters)
{
    while(1)
    {
        printf("消费者: 等待信号量...\n");
        
        // 等待信号量，无限期等待
        if(xSemaphoreTake(xBinarySemaphore, portMAX_DELAY) == pdTRUE)
        {
            printf("消费者: 收到信号量，开始处理数据...\n");
            
            // 模拟数据处理
            vTaskDelay(pdMS_TO_TICKS(500)); // 延时500ms
            
            printf("消费者: 数据处理完成！\n\n");
        }
    }
}

// 主函数
int main(void)
{
    printf("=== FreeRTOS 信号量示例程序 ===\n\n");
    
    // 创建二进制信号量
    xBinarySemaphore = xSemaphoreCreateBinary();
    
    if(xBinarySemaphore == NULL)
    {
        printf("错误: 信号量创建失败！\n");
        return -1;
    }
    
    printf("信号量创建成功！\n\n");
    
    // 创建生产者任务
    if(xTaskCreate(vProducerTask,           // 任务函数
                   "Producer",              // 任务名称
                   1000,                    // 堆栈大小
                   NULL,                    // 任务参数
                   2,                       // 任务优先级
                   NULL) != pdPASS)         // 任务句柄
    {
        printf("错误: 生产者任务创建失败！\n");
        return -1;
    }
    
    // 创建消费者任务
    if(xTaskCreate(vConsumerTask,           // 任务函数
                   "Consumer",              // 任务名称
                   1000,                    // 堆栈大小
                   NULL,                    // 任务参数
                   1,                       // 任务优先级
                   NULL) != pdPASS)         // 任务句柄
    {
        printf("错误: 消费者任务创建失败！\n");
        return -1;
    }
    
    printf("任务创建成功，启动调度器...\n\n");
    
    // 启动任务调度器
    vTaskStartScheduler();
    
    // 如果程序运行到这里，说明内存不足
    printf("错误: 调度器启动失败，内存不足！\n");
    return -1;
}

// FreeRTOS钩子函数 - 当没有足够内存创建空闲任务时调用
void vApplicationMallocFailedHook(void)
{
    printf("内存分配失败！\n");
    while(1);
}

// FreeRTOS钩子函数 - 堆栈溢出检测
void vApplicationStackOverflowHook(TaskHandle_t xTask, char *pcTaskName)
{
    printf("任务 %s 堆栈溢出！\n", pcTaskName);
    while(1);
}
```

# 主要特性

1. 信号量创建：使用 xSemaphoreCreateBinary() 创建二进制信号量
2. 信号量操作：

    * xSemaphoreGive()：释放信号量（生产者使用）
    * xSemaphoreTake()：获取信号量（消费者使用）


3. 任务同步：消费者任务会阻塞等待，直到生产者发送信号量
4. 错误处理：包含信号量创建失败和任务创建失败的检查

# 运行流程

1. 程序启动后创建信号量和两个任务
2. 消费者任务开始等待信号量
3. 生产者任务每2秒发送一个信号量
4. 消费者收到信号量后处理数据（模拟500ms处理时间）
5. 循环重复此过程
