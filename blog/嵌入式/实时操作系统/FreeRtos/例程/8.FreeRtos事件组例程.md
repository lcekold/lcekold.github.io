# 举例说明

## 场景

创建两个任务：一个模拟温度传感器任务，一个模拟湿度传感器任务。它们分别独立地采集数据，采集完成后设置事件组中的不同标志位。一个数据处理任务等待这两个标志位都被设置后，开始执行处理操作。

## 代码示例

```c
#include "FreeRTOS.h"
#include "task.h"
#include "event_groups.h"
#include "stm32f1xx_hal.h"  // 根据你的STM32型号调整
#include <stdio.h>
#include <stdlib.h>

// 定义事件标志位
#define TEMP_DATA_READY_BIT    (1 << 0)  // 温度数据就绪
#define HUMI_DATA_READY_BIT    (1 << 1)  // 湿度数据就绪
#define ALL_DATA_READY_BITS    (TEMP_DATA_READY_BIT | HUMI_DATA_READY_BIT)

// 事件组句柄
EventGroupHandle_t xSensorEventGroup;

// 任务句柄
TaskHandle_t xTempTaskHandle;
TaskHandle_t xHumiTaskHandle;
TaskHandle_t xDataProcessTaskHandle;

// 全局变量存储传感器数据
volatile float g_temperature = 0.0f;
volatile float g_humidity = 0.0f;

// 温度传感器任务
void vTemperatureSensorTask(void *pvParameters)
{
    TickType_t xLastWakeTime;
    const TickType_t xSamplePeriod = pdMS_TO_TICKS(2000); // 2秒采样一次
    
    xLastWakeTime = xTaskGetTickCount();
    
    printf("温度传感器任务启动\r\n");
    
    while(1)
    {
        printf("温度传感器：开始采集数据...\r\n");
        
        // 模拟温度传感器数据采集过程（需要一定时间）
        vTaskDelay(pdMS_TO_TICKS(500)); // 模拟采集延时500ms
        
        // 模拟读取温度值（15-35度范围）
        g_temperature = 15.0f + (rand() % 200) / 10.0f;
        
        printf("温度传感器：采集完成，温度 = %.1f°C\r\n", g_temperature);
        
        // 设置温度数据就绪标志位
        xEventGroupSetBits(xSensorEventGroup, TEMP_DATA_READY_BIT);
        
        // 等待下一次采样周期
        vTaskDelayUntil(&xLastWakeTime, xSamplePeriod);
    }
}

// 湿度传感器任务
void vHumiditySensorTask(void *pvParameters)
{
    TickType_t xLastWakeTime;
    const TickType_t xSamplePeriod = pdMS_TO_TICKS(1500); // 1.5秒采样一次
    
    xLastWakeTime = xTaskGetTickCount();
    
    printf("湿度传感器任务启动\r\n");
    
    while(1)
    {
        printf("湿度传感器：开始采集数据...\r\n");
        
        // 模拟湿度传感器数据采集过程（需要一定时间）
        vTaskDelay(pdMS_TO_TICKS(300)); // 模拟采集延时300ms
        
        // 模拟读取湿度值（30-90%范围）
        g_humidity = 30.0f + (rand() % 600) / 10.0f;
        
        printf("湿度传感器：采集完成，湿度 = %.1f%%\r\n", g_humidity);
        
        // 设置湿度数据就绪标志位
        xEventGroupSetBits(xSensorEventGroup, HUMI_DATA_READY_BIT);
        
        // 等待下一次采样周期
        vTaskDelayUntil(&xLastWakeTime, xSamplePeriod);
    }
}

// 数据处理任务
void vDataProcessTask(void *pvParameters)
{
    EventBits_t xEventBits;
    const TickType_t xTimeoutTicks = pdMS_TO_TICKS(10000); // 最多等待10秒
    float local_temp, local_humi;
    float heat_index; // 体感温度
    
    printf("数据处理任务启动\r\n");
    
    while(1)
    {
        printf("\r\n数据处理任务：等待温度和湿度数据就绪...\r\n");
        
        // 等待温度和湿度数据都就绪
        xEventBits = xEventGroupWaitBits(
            xSensorEventGroup,      // 事件组句柄
            ALL_DATA_READY_BITS,    // 等待的事件位
            pdTRUE,                 // 退出前清除事件位
            pdTRUE,                 // 等待所有位都被设置（AND操作）
            xTimeoutTicks           // 超时时间
        );
        
        // 检查是否所有数据都已就绪
        if((xEventBits & ALL_DATA_READY_BITS) == ALL_DATA_READY_BITS)
        {
            printf("数据处理任务：所有传感器数据已就绪，开始处理...\r\n");
            
            // 获取传感器数据（临界区保护）
            taskENTER_CRITICAL();
            local_temp = g_temperature;
            local_humi = g_humidity;
            taskEXIT_CRITICAL();
            
            // 模拟数据处理过程
            vTaskDelay(pdMS_TO_TICKS(200)); // 处理延时
            
            // 计算体感温度（简化公式）
            heat_index = local_temp + 0.4f * (local_humi - 50.0f);
            
            printf("==================== 数据处理结果 ====================\r\n");
            printf("温度：%.1f°C\r\n", local_temp);
            printf("湿度：%.1f%%\r\n", local_humi);
            printf("体感温度：%.1f°C\r\n", heat_index);
            
            // 环境评估
            if(heat_index < 18.0f)
            {
                printf("环境评估：偏冷\r\n");
            }
            else if(heat_index < 26.0f)
            {
                printf("环境评估：舒适\r\n");
            }
            else if(heat_index < 32.0f)
            {
                printf("环境评估：偏热\r\n");
            }
            else
            {
                printf("环境评估：过热\r\n");
            }
            
            printf("==================================================\r\n\r\n");
            
            // 可以在这里添加其他处理逻辑，如：
            // - 数据存储
            // - 网络上传
            // - 控制风扇、加湿器等设备
        }
        else
        {
            printf("数据处理任务：等待数据超时！\r\n");
            
            // 检查具体哪个传感器数据缺失
            if(!(xEventBits & TEMP_DATA_READY_BIT))
            {
                printf("警告：温度传感器数据未就绪\r\n");
            }
            if(!(xEventBits & HUMI_DATA_READY_BIT))
            {
                printf("警告：湿度传感器数据未就绪\r\n");
            }
        }
        
        // 短暂延时后继续下一轮处理
        vTaskDelay(pdMS_TO_TICKS(100));
    }
}

// 初始化函数
void SensorEventGroupExample_Init(void)
{
    printf("正在初始化传感器事件组例程...\r\n");
    
    // 创建事件组
    xSensorEventGroup = xEventGroupCreate();
    
    if(xSensorEventGroup != NULL)
    {
        printf("事件组创建成功\r\n");
        
        // 创建温度传感器任务
        if(xTaskCreate(vTemperatureSensorTask, "TempTask", 256, NULL, 2, &xTempTaskHandle) != pdPASS)
        {
            printf("温度传感器任务创建失败\r\n");
            return;
        }
        
        // 创建湿度传感器任务
        if(xTaskCreate(vHumiditySensorTask, "HumiTask", 256, NULL, 2, &xHumiTaskHandle) != pdPASS)
        {
            printf("湿度传感器任务创建失败\r\n");
            return;
        }
        
        // 创建数据处理任务
        if(xTaskCreate(vDataProcessTask, "ProcessTask", 512, NULL, 1, &xDataProcessTaskHandle) != pdPASS)
        {
            printf("数据处理任务创建失败\r\n");
            return;
        }
        
        printf("所有任务创建成功，系统开始运行\r\n");
    }
    else
    {
        printf("事件组创建失败\r\n");
    }
}

// 清理资源函数（可选）
void SensorEventGroup_Cleanup(void)
{
    // 删除任务
    if(xTempTaskHandle != NULL)
    {
        vTaskDelete(xTempTaskHandle);
        xTempTaskHandle = NULL;
    }
    
    if(xHumiTaskHandle != NULL)
    {
        vTaskDelete(xHumiTaskHandle);
        xHumiTaskHandle = NULL;
    }
    
    if(xDataProcessTaskHandle != NULL)
    {
        vTaskDelete(xDataProcessTaskHandle);
        xDataProcessTaskHandle = NULL;
    }
    
    // 删除事件组
    if(xSensorEventGroup != NULL)
    {
        vEventGroupDelete(xSensorEventGroup);
        xSensorEventGroup = NULL;
    }
    
    printf("资源清理完成\r\n");
}
```

## 主要功能

主要功能
1. 温度传感器任务

    * 每2秒采集一次温度数据（15-35°C范围）
    * 模拟500ms的采集延时
    * 采集完成后设置TEMP_DATA_READY_BIT标志位

2. 湿度传感器任务

    * 每1.5秒采集一次湿度数据（30-90%范围）
    * 模拟300ms的采集延时
    * 采集完成后设置HUMI_DATA_READY_BIT标志位

3. 数据处理任务

    * 等待温度和湿度数据都就绪（AND逻辑）
    * 计算体感温度并进行环境评估
    * 提供超时处理和错误检测

## 关键技术点

* 事件同步：使用xEventGroupWaitBits()等待多个事件
* 临界区保护：使用taskENTER_CRITICAL()保护共享数据
* 周期性任务：使用vTaskDelayUntil()实现精确的采样周期
* 超时处理：10秒超时机制防止系统阻塞

## 使用方法

1. 在main函数中调用SensorEventGroupExample_Init()
2. 确保FreeRTOSConfig.h中启用相关配置：

        #define configUSE_16_BIT_TICKS 0
        #define INCLUDE_xEventGroupSetBitsFromISR 1

## 运行效果
系统会持续显示：

* 各传感器的采集过程
* 数据处理结果（温度、湿度、体感温度）
* 环境评估（偏冷/舒适/偏热/过热）
* 异常情况的处理
