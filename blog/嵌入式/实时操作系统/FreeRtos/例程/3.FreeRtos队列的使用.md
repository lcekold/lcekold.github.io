# 队列例程

## 示例概述
* 创建两个队列：一个用于红外遥控器的按键数据，另一个用于旋转编码器的旋转数据。

* 两个发送任务（模拟中断）：分别模拟红外遥控器和旋转编码器的输入，将数据发送到各自的队列中。

* 一个接收任务：等待并读取这两个队列中的数据，根据接收到的数据执行相应的操作。

## 代码实现

```c
#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/queue.h"

// 定义红外遥控器的按键数据类型
typedef struct {
    uint8_t key_code; // 按键键值
} ir_data_t;

// 定义旋转编码器的数据类型
typedef struct {
    int8_t direction; // 旋转方向：1表示顺时针，-1表示逆时针
    int16_t speed;    // 旋转速度（模拟值）
} rotary_encoder_data_t;

// 队列句柄
QueueHandle_t xQueueIR;     // 红外遥控器队列
QueueHandle_t xQueueRotary; // 旋转编码器队列

// 模拟红外遥控器发送任务（模拟中断服务程序通过队列发送数据）
void vIRSenderTask(void *pvParameters) {
    ir_data_t ir_data;
    uint8_t key_value = 0;

    while (1) {
        // 模拟不同的按键按下，这里简单循环键值
        key_value = (key_value % 4) + 1; // 键值从1循环到4
        ir_data.key_code = key_value;

        // 发送数据到红外队列，阻塞时间为0，如果队列满则立即返回
        if (xQueueSend(xQueueIR, &ir_data, 0) != pdPASS) {
            printf("IR Queue full, data lost: %d\n", ir_data.key_code);
        } else {
            printf("IR Key sent: %d\n", ir_data.key_code);
        }

        vTaskDelay(pdMS_TO_TICKS(1000)); // 每隔1000ms发送一次，模拟按键间隔
    }
}

// 模拟旋转编码器发送任务（模拟中断服务程序通过队列发送数据）
void vRotaryEncoderSenderTask(void *pvParameters) {
    rotary_encoder_data_t rotary_data;

    while (1) {
        // 模拟旋转编码器动作，交替顺时针和逆时针旋转，速度随机模拟
        rotary_data.direction = (rotary_data.direction == 1) ? -1 : 1; // 方向交替变化
        rotary_data.speed = (rand() % 100) + 1; // 速度1-100之间的随机值

        // 发送数据到旋转编码器队列，阻塞时间为0
        if (xQueueSend(xQueueRotary, &rotary_data, 0) != pdPASS) {
            printf("Rotary Queue full, data lost. Dir: %d, Speed: %d\n", rotary_data.direction, rotary_data.speed);
        } else {
            printf("Rotary Data sent: Direction %d, Speed %d\n", rotary_data.direction, rotary_data.speed);
        }

        vTaskDelay(pdMS_TO_TICKS(500)); // 每隔500ms发送一次，模拟旋转间隔
    }
}

// 综合处理任务（从两个队列中接收数据并处理）
void vInputProcessorTask(void *pvParameters) {
    ir_data_t received_ir_data;
    rotary_encoder_data_t received_rotary_data;
    QueueHandle_t xQueue;
    BaseType_t xResult;

    while (1) {
        // 等待任意一个队列有数据到达，阻塞等待
        xResult = xQueueReceive(xQueueIR, &received_ir_data, portMAX_DELAY); // 永久阻塞直到收到数据

        if (xResult == pdPASS) {
            // 处理红外遥控器数据
            printf("Processing IR Key: %d. Action: ", received_ir_data.key_code);
            switch (received_ir_data.key_code) {
                case 1:
                    printf("Move LEFT.\n");
                    break;
                case 2:
                    printf("Move RIGHT.\n");
                    break;
                case 3:
                    printf("SELECT.\n");
                    break;
                case 4:
                    printf("BACK.\n");
                    break;
                default:
                    printf("Unknown key.\n");
                    break;
            }
        }

        // 非阻塞地检查旋转编码器队列，避免因为一直等待旋转编码器数据而无法处理后续的红外数据
        xResult = xQueueReceive(xQueueRotary, &received_rotary_data, 0); // 不阻塞

        if (xResult == pdPASS) {
            // 处理旋转编码器数据
            printf("Processing Rotary Encoder: Direction %s, Speed %d. Action: %s by %d units.\n",
                   (received_rotary_data.direction == 1) ? "CW" : "CCW",
                   received_rotary_data.speed,
                   (received_rotary_data.direction == 1) ? "Increase" : "Decrease",
                   received_rotary_data.speed);
        }
    }
}

void app_main() {
    // 创建队列
    // 红外队列：最多可存储5个ir_data_t元素
    xQueueIR = xQueueCreate(5, sizeof(ir_data_t));
    // 旋转编码器队列：最多可存储5个rotary_encoder_data_t元素
    xQueueRotary = xQueueCreate(5, sizeof(rotary_encoder_data_t));

    if (xQueueIR == NULL || xQueueRotary == NULL) {
        printf("Queue creation failed!.\n");
        return;
    }

    // 创建任务
    xTaskCreate(vIRSenderTask, "IR Sender", 2048, NULL, 1, NULL);
    xTaskCreate(vRotaryEncoderSenderTask, "Rotary Sender", 2048, NULL, 1, NULL);
    xTaskCreate(vInputProcessorTask, "Input Processor", 2048, NULL, 2, NULL); // 赋予较高的优先级2，确保及时处理输入

    printf("Multi-input device example started.\n");
}
```

## 关键点解释

* 队列创建：使用 xQueueCreate 创建了两个队列，分别用于红外和旋转编码器数据。

* 数据发送任务：vIRSenderTask 和 vRotaryEncoderSenderTask 任务模拟了两个输入设备的行为，定期生成数据并发送到各自的队列中。它们使用 xQueueSend 并设置阻塞时间为 0（0 在 xQueueSend 中表示如果队列满就立即返回错误而不阻塞），模拟了中断服务程序中常用的 xQueueSendFromISR 的非阻塞特性。

* 数据接收与处理任务：vInputProcessorTask 任务负责处理来自两个队列的输入。它使用 xQueueReceive 等待红外队列的数据（portMAX_DELAY 表示无限期阻塞直到有数据），然后非阻塞地检查旋转编码器队列（超时时间设为 0），以免错过频繁的红外信号。在实际应用中，你可能需要更复杂的机制，如队列集（Queue Sets） 来同时等待多个队列。

* 优先级：处理任务的优先级（2）设置得比发送任务（1）高，以确保输入数据能被及时处理，避免队列溢出。

## 运行预期

运行此代码，你将在串口监视器中看到类似以下的输出（具体数值和顺序会因随机数和任务调度而异）：

# 队列集例程

## 场景描述

假设我们有一个嵌入式设备，有两个输入源：

1. 按键：用户输入，重要性高，需要立即响应。

2. 温度传感器：周期性读取，重要性较低，可以稍后处理。

不使用队列集的问题：如果在一个任务中循环查询这两个队列，当温度传感器队列为空时，任务会阻塞在读取温度队列上，导致按键事件无法被及时响应。

使用队列集的改善：创建一个队列集，将两个队列都加入其中。处理任务只需等待这个队列集。任何一个队列有数据时，任务都会立即被唤醒，从而实现对不同优先级事件的即时、统一处理。

## 代码实现

### 1. 头文件包含和定义

```c
#include <stdio.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "freertos/queue.h"

// 定义消息类型
typedef struct {
    uint8_t key_id;     // 按键ID
    bool is_pressed;    // 按下或释放
} key_event_t;

typedef struct {
    float temperature;  // 温度值
    float humidity;     // 湿度值
} sensor_data_t;

// 全局句柄
QueueHandle_t xKeyQueue;        // 按键队列
QueueHandle_t xSensorQueue;     // 传感器队列
QueueSetHandle_t xInputQueueSet; // 队列集
```

### 2. 模拟生产者任务（代替实际中断）

```c
// 模拟按键中断服务程序（实际应用中应使用 xQueueSendFromISR）
void vKeyProducerTask(void *pvParameters) {
    key_event_t key_event = {1, true};
    uint8_t key_counter = 0;

    while (1) {
        // 模拟按键按下事件，键值在1-3之间循环
        key_event.key_id = (key_counter % 3) + 1;
        key_event.is_pressed = true;
        
        if (xQueueSend(xKeyQueue, &key_event, 0) != pdPASS) {
            printf("[Key] Queue full, event lost: Key%d\n", key_event.key_id);
        } else {
            printf("[Key] Event sent: Key%d %s\n", key_event.key_id, 
                   key_event.is_pressed ? "PRESSED" : "RELEASED");
        }
        
        key_counter++;
        vTaskDelay(pdMS_TO_TICKS(1500)); // 每1.5秒模拟一次按键
    }
}

// 模拟传感器数据采集任务
void vSensorProducerTask(void *pvParameters) {
    sensor_data_t sensor_data = {25.0, 50.0};

    while (1) {
        // 模拟传感器读数（略有变化）
        sensor_data.temperature += (rand() % 5) - 2; // -2°C 到 +2°C 的变化
        sensor_data.humidity += (rand() % 10) - 5;   // -5% 到 +5% 的变化
        
        if (xQueueSend(xSensorQueue, &sensor_data, 0) != pdPASS) {
            printf("[Sensor] Queue full, data lost: %.1f°C\n", sensor_data.temperature);
        } else {
            printf("[Sensor] Data sent: %.1f°C, %.1f%%\n", 
                   sensor_data.temperature, sensor_data.humidity);
        }
        
        vTaskDelay(pdMS_TO_TICKS(3000)); // 每3秒采集一次
    }
}
```

### 3. 使用队列集的消费者任务（核心改善）

```c
void vInputProcessorTask(void *pvParameters) {
    key_event_t key_event;
    sensor_data_t sensor_data;
    QueueHandle_t xActivatedQueue;
    
    printf("Input Processor started. Waiting for events...\n");

    while (1) {
        // 核心改善：阻塞等待队列集中的任意队列有数据
        // 无论是按键还是传感器数据，都会立即唤醒此任务
        xActivatedQueue = (QueueHandle_t) xQueueSelectFromSet(xInputQueueSet, portMAX_DELAY);
        
        // 判断是哪个队列被激活并处理相应数据
        if (xActivatedQueue == xKeyQueue) {
            // 处理按键事件（高优先级）
            if (xQueueReceive(xKeyQueue, &key_event, 0) == pdPASS) {
                printf(">>> PROCESSING KEY: Key%d %s - Immediate response!\n",
                       key_event.key_id, key_event.is_pressed ? "PRESSED" : "RELEASED");
                
                // 这里可以执行紧急操作，如关机、确认等
                if (key_event.key_id == 1 && key_event.is_pressed) {
                    printf("    --> Emergency stop initiated!\n");
                }
            }
        } 
        else if (xActivatedQueue == xSensorQueue) {
            // 处理传感器数据（低优先级）
            if (xQueueReceive(xSensorQueue, &sensor_data, 0) == pdPASS) {
                printf(">>> PROCESSING SENSOR: %.1f°C, %.1f%% - Logging data\n",
                       sensor_data.temperature, sensor_data.humidity);
                
                // 这里可以执行数据记录、显示更新等非紧急操作
                if (sensor_data.temperature > 30.0) {
                    printf("    --> Warning: Temperature too high!\n");
                }
            }
        }
        
        // 处理完一个事件后，可以快速返回继续等待下一个事件
    }
}
```

### 4. 初始化函数

```c
void app_main() {
    // 1. 创建单个队列
    xKeyQueue = xQueueCreate(5, sizeof(key_event_t));     // 按键队列，深度5
    xSensorQueue = xQueueCreate(3, sizeof(sensor_data_t)); // 传感器队列，深度3

    // 2. 创建队列集（改善的核心）
    // 参数是队列集能容纳的总项目数（所有成员队列长度之和）
    xInputQueueSet = xQueueCreateSet(5 + 3);
    
    // 3. 将队列添加到队列集中
    xQueueAddToSet(xKeyQueue, xInputQueueSet);
    xQueueAddToSet(xSensorQueue, xInputQueueSet);

    // 4. 创建任务
    xTaskCreate(vKeyProducerTask, "Key Producer", 2048, NULL, 1, NULL);
    xTaskCreate(vSensorProducerTask, "Sensor Producer", 2048, NULL, 1, NULL);
    xTaskCreate(vInputProcessorTask, "Input Processor", 2048, NULL, 2, NULL); // 较高优先级

    printf("Queue Set example started. Framework improved!\n");
}
```

## 框架改善的核心优势

1. 即时响应：按键事件不再需要等待传感器读取周期结束，实现了真正的即时响应。

2. 资源高效：处理任务在无事件时完全阻塞，不占用 CPU 时间。

3. 模块化清晰：不同的输入源有独立的队列，便于维护和扩展。

4. 易于扩展：如果要添加第三个输入设备（如网络数据），只需：

        QueueHandle_t xNetworkQueue = xQueueCreate(...);
        xQueueAddToSet(xNetworkQueue, xInputQueueSet);
        // 在处理任务中添加相应的判断分支

5. 优先级管理：在处理任务中可以根据事件类型实现软件优先级（先处理按键，后处理传感器）。

这个例子展示了如何用队列集将程序框架从"主动轮询"改善为"事件驱动"，这是嵌入式系统设计中一个非常重要的架构优化。
