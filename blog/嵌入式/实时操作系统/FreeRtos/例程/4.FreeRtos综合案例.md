# 例程

## 功能描述

我现在使用ESP32，使用FreeRtos实现以下功能：
1. 实现与上位机的串口数据收发
2. 从dht11和mq-5获取温湿度数据和气体质量数据，并显示在OLED屏上
3. 能够接收上位机的数据从而控制灯光亮灭、风扇的开启

请问使用FreeRtos应该如何实现，我应该使用FreeRtos的哪些机制？


## 系统架构设计

### 任务划分

1. 串口通信任务 - 处理与上位机的数据收发

2. 传感器读取任务 - 从DHT11和MQ-5获取数据

3. 显示任务 - 将数据展示在OLED屏幕上

4. 控制任务 - 处理灯光和风扇控制

5. 主监控任务 - 协调各任务(可选)

### FreeRTOS机制应用

1. 任务(Task) - 每个功能独立任务

2. 队列(Queue) - 任务间数据传输

3. 信号量(Semaphore) - 资源访问控制

4. 事件组(Event Group) - 任务同步

## 代码实现框架

```c
#include <freertos/FreeRTOS.h>
#include <freertos/task.h>
#include <freertos/queue.h>
#include <freertos/semphr.h>
#include <freertos/event_groups.h>
#include <Wire.h>
#include <Adafruit_SSD1306.h>
#include "DHT.h"

// 硬件定义
#define DHTPIN 4
#define DHTTYPE DHT11
#define MQ5_PIN 34
#define LED_PIN 2
#define FAN_PIN 15

// 事件组标志位定义
#define SENSOR_READ_EVENT (1 << 0)
#define UART_RX_EVENT (1 << 1)
#define DISPLAY_UPDATE_EVENT (1 << 2)

// 全局变量和对象声明
DHT dht(DHTPIN, DHTTYPE);
Adafruit_SSD1306 display(128, 64, &Wire, -1);
EventGroupHandle_t xEventGroup;
QueueHandle_t xSensorQueue;
QueueHandle_t xUartQueue;
SemaphoreHandle_t xI2CSemaphore;

// 数据结构定义
typedef struct {
  float temperature;
  float humidity;
  int gasValue;
} SensorData_t;

typedef struct {
  char command[20];
} UartCommand_t;

// 任务函数声明
void vUartTask(void *pvParameters);
void vSensorTask(void *pvParameters);
void vDisplayTask(void *pvParameters);
void vControlTask(void *pvParameters);

void setup() {
  Serial.begin(115200);
  
  // 初始化硬件
  pinMode(LED_PIN, OUTPUT);
  pinMode(FAN_PIN, OUTPUT);
  dht.begin();
  
  // 初始化OLED
  Wire.begin();
  display.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0,0);
  display.println("System Starting");
  display.display();
  
  // 创建FreeRTOS对象
  xEventGroup = xEventGroupCreate();
  xSensorQueue = xQueueCreate(5, sizeof(SensorData_t));
  xUartQueue = xQueueCreate(5, sizeof(UartCommand_t));
  xI2CSemaphore = xSemaphoreCreateMutex();
  
  // 创建任务
  xTaskCreate(vUartTask, "Uart Task", 2048, NULL, 3, NULL);
  xTaskCreate(vSensorTask, "Sensor Task", 2048, NULL, 2, NULL);
  xTaskCreate(vDisplayTask, "Display Task", 2048, NULL, 1, NULL);
  xTaskCreate(vControlTask, "Control Task", 2048, NULL, 2, NULL);
}

void loop() {
  // FreeRTOS调度器接管后，loop函数通常为空
  vTaskDelete(NULL);
}

// 串口通信任务
void vUartTask(void *pvParameters) {
  UartCommand_t receivedCommand;
  
  for(;;) {
    if(Serial.available() > 0) {
      String input = Serial.readStringUntil('\n');
      input.trim();
      
      if(input.length() > 0) {
        // 解析命令
        if(input.startsWith("LED")) {
          strncpy(receivedCommand.command, input.c_str(), sizeof(receivedCommand.command));
          xQueueSend(xUartQueue, &receivedCommand, portMAX_DELAY);
        }
        else if(input.startsWith("FAN")) {
          strncpy(receivedCommand.command, input.c_str(), sizeof(receivedCommand.command));
          xQueueSend(xUartQueue, &receivedCommand, portMAX_DELAY);
        }
        else if(input.equals("GET_DATA")) {
          // 触发传感器读取
          xEventGroupSetBits(xEventGroup, SENSOR_READ_EVENT);
        }
      }
    }
    
    vTaskDelay(100 / portTICK_PERIOD_MS);
  }
}

// 传感器读取任务
void vSensorTask(void *pvParameters) {
  SensorData_t sensorData;
  
  for(;;) {
    // 等待读取事件或定时读取
    EventBits_t uxBits = xEventGroupWaitBits(
      xEventGroup, 
      SENSOR_READ_EVENT, 
      pdTRUE, 
      pdFALSE, 
      pdMS_TO_TICKS(5000) // 5秒超时，实现定时读取
    );
    
    if((uxBits & SENSOR_READ_EVENT) != 0 || uxBits == 0) {
      // 读取DHT11数据
      sensorData.humidity = dht.readHumidity();
      sensorData.temperature = dht.readTemperature();
      
      // 读取MQ-5数据
      sensorData.gasValue = analogRead(MQ5_PIN);
      
      // 发送到显示队列
      xQueueSend(xSensorQueue, &sensorData, portMAX_DELAY);
      
      // 设置显示更新事件
      xEventGroupSetBits(xEventGroup, DISPLAY_UPDATE_EVENT);
      
      // 向上位机发送数据
      Serial.print("T:");
      Serial.print(sensorData.temperature);
      Serial.print(" H:");
      Serial.print(sensorData.humidity);
      Serial.print(" G:");
      Serial.println(sensorData.gasValue);
    }
    
    vTaskDelay(100 / portTICK_PERIOD_MS);
  }
}

// 显示任务
void vDisplayTask(void *pvParameters) {
  SensorData_t sensorData;
  
  for(;;) {
    // 等待显示更新事件
    EventBits_t uxBits = xEventGroupWaitBits(
      xEventGroup, 
      DISPLAY_UPDATE_EVENT, 
      pdTRUE, 
      pdFALSE, 
      portMAX_DELAY
    );
    
    if((uxBits & DISPLAY_UPDATE_EVENT) != 0) {
      // 从队列获取传感器数据
      if(xQueueReceive(xSensorQueue, &sensorData, portMAX_DELAY) == pdPASS) {
        // 获取I2C访问权
        if(xSemaphoreTake(xI2CSemaphore, portMAX_DELAY) == pdTRUE) {
          display.clearDisplay();
          display.setCursor(0,0);
          display.print("Temp: ");
          display.print(sensorData.temperature);
          display.println(" C");
          display.print("Humidity: ");
          display.print(sensorData.humidity);
          display.println(" %");
          display.print("Gas: ");
          display.println(sensorData.gasValue);
          display.display();
          
          // 释放I2C访问权
          xSemaphoreGive(xI2CSemaphore);
        }
      }
    }
    
    vTaskDelay(100 / portTICK_PERIOD_MS);
  }
}

// 控制任务
void vControlTask(void *pvParameters) {
  UartCommand_t command;
  
  for(;;) {
    // 从队列获取控制命令
    if(xQueueReceive(xUartQueue, &command, portMAX_DELAY) == pdPASS) {
      if(strncmp(command.command, "LED ON", 6) == 0) {
        digitalWrite(LED_PIN, HIGH);
        Serial.println("LED turned ON");
      }
      else if(strncmp(command.command, "LED OFF", 7) == 0) {
        digitalWrite(LED_PIN, LOW);
        Serial.println("LED turned OFF");
      }
      else if(strncmp(command.command, "FAN ON", 6) == 0) {
        digitalWrite(FAN_PIN, HIGH);
        Serial.println("FAN turned ON");
      }
      else if(strncmp(command.command, "FAN OFF", 7) == 0) {
        digitalWrite(FAN_PIN, LOW);
        Serial.println("FAN turned OFF");
      }
    }
    
    vTaskDelay(50 / portTICK_PERIOD_MS);
  }
}
```

## FreeRTOS机制说明

1. 任务(Tasks)

    * 将系统功能划分为独立任务，每个任务有明确职责

    * 设置不同优先级确保关键任务及时响应

2. 队列(Queues)

    * 传感器数据队列：传输传感器读数到显示任务

    * 串口命令队列：传输控制命令到控制任务

3. 事件组(Event Groups)

    * 用于任务间同步和事件通知

    * 传感器读取事件、显示更新事件等

4. 信号量(Semaphores)

    * I2C互斥信号量：防止OLED和传感器同时访问I2C总线

## 使用说明

1. 上传代码到ESP32开发板

2. 连接硬件：

* DHT11 -> GPIO4

* MQ-5 -> GPIO34

* LED -> GPIO2

* 风扇 -> GPIO15

* OLED -> I2C接口

3. 通过串口发送命令：

* "LED ON"/"LED OFF" - 控制灯光

* "FAN ON"/"FAN OFF" - 控制风扇

* "GET_DATA" - 主动获取传感器数据
